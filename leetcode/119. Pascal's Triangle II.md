// problem :

Given an integer rowIndex, return the rowIndexth (0-indexed) row of the Pascal's triangle.

In Pascal's triangle, each number is the sum of the two numbers directly above it as shown:
 

Example 1:

Input: rowIndex = 3
Output: [1,3,3,1]

Example 2:

Input: rowIndex = 0
Output: [1]

Example 3:

Input: rowIndex = 1
Output: [1,1]


// SOLUTION :

```cpp
class Solution {
public:
    vector<int> getRow(int rowIndex) {
        // Start with the first row of Pascal's Triangle = [1]
        vector<int> row(1, 1);

        // Build row by row until reaching rowIndex
        for (int i = 0; i < rowIndex; i++) {
            vector<int> newRow;
            
            // Every row starts with 1
            newRow.push_back(1);

            // Middle elements = sum of two adjacent elements from previous row
            // Example: if row = [1, 3, 3, 1], then middle = [4, 6, 4]
            for (int j = 1; j < row.size(); j++) {
                newRow.push_back(row[j - 1] + row[j]);
            }

            // Every row ends with 1
            newRow.push_back(1);

            // Move to the newly built row
            row = newRow;
        }

        // Return the rowIndex-th row (0-based index)
        return row;        
    }
};

```
