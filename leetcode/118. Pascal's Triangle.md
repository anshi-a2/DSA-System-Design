// PROBLEM :

Given an integer numRows, return the first numRows of Pascal's triangle.

In Pascal's triangle, each number is the sum of the two numbers directly above it as shown:
 

Example 1:

Input: numRows = 5
Output: [[1],[1,1],[1,2,1],[1,3,3,1],[1,4,6,4,1]]
Example 2:

Input: numRows = 1
Output: [[1]]

// SOLUTION :

```cpp
class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        // Result container: will store all rows of Pascal's Triangle
        vector<vector<int>> res;

        // First row is always [1]
        res.push_back({1});

        // Build the triangle row by row
        for (int i = 0; i < numRows - 1; i++) {
            // Example: if last row = [1, 2, 1], dummyRow = [0, 1, 2, 1, 0]
            vector<int> dummyRow = {0};
            dummyRow.insert(dummyRow.end(), res.back().begin(), res.back().end());
            dummyRow.push_back(0);

            // New row to build
            vector<int> row;

            // Each element is the sum of two adjacent elements from dummyRow
            // Example: [0, 1, 2, 1, 0] → [1, 3, 3, 1]
            for (int j = 0; j < dummyRow.size() - 1; j++) {
                row.push_back(dummyRow[j] + dummyRow[j + 1]);
            }

            // Add the new row to result
            res.push_back(row);
        }

        return res;        
    }
};

```
